local dragging = false
local dragStart
local startPos

toggleButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = toggleButton.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart

        toggleButton.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

local function createPart(parent, name, size, color, material, transparency)
    local part = Instance.new("Part")
    part.Name = name
    part.Size = size
    part.Color = color
    part.Material = material or Enum.Material.Neon
    part.Transparency = transparency or 0
    part.Anchored = true
    part.CanCollide = false
    part.CanTouch = false
    part.CanQuery = false
    part.CastShadow = true
    part.Parent = parent
    return part
end

local function getFixedMonsterPosition()
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")

    if root then
        return root.Position + fixedMonsterOffset
    end

    return Vector3.new(0, 55, -350)
end

local function destroyMonster()
    monsterParts = {}
    monsterBaseCFrames = {}
    monsterWorldPosition = nil

    if monster then
        monster:Destroy()
        monster = nil
    end

    monsterLight = nil
    monsterHighlight = nil
end

local function updateFloorList()
    floorParts = {}
    originalColors = {}
    originalCFrames = {}

    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local size = obj.Size
            local isHorizontal = size.Y < size.X and size.Y < size.Z
            local isLargeArea = (size.X >= 10 and size.Z >= 10) or obj.Position.Y < 5

            if isHorizontal or isLargeArea then
                table.insert(floorParts, obj)
                originalColors[obj] = obj.Color
                originalCFrames[obj] = obj.CFrame
            end
        end
    end
end

local function prepareConstructionSwirl()
    swirlParts = {}
    swirlData = {}
    swirlOriginalCFrames = {}

    if not monsterWorldPosition then
        return
    end

    for _, obj in ipairs(floorParts) do
        if obj and obj.Parent and not obj:IsDescendantOf(monster) then
            local distance = (obj.Position - monsterWorldPosition).Magnitude

            if distance <= 850 then
                table.insert(swirlParts, obj)
                swirlOriginalCFrames[obj] = obj.CFrame

                local relative = obj.Position - monsterWorldPosition
                local radius = math.clamp(Vector3.new(relative.X, 0, relative.Z).Magnitude, 45, 420)
                local angle = math.atan2(relative.Z, relative.X)

                swirlData[obj] = {
                    radius = radius,
                    angle = angle,
                    height = math.clamp(relative.Y, -100, 220),
                    speed = math.random(45, 100) / 100,
                    phase = math.random() * math.pi * 2,
                    spin = math.random(-100, 100) / 100
                }
            end
        end
    end

    constructionSwirlActive = true
end

local function stopConstructionSwirl()
    constructionSwirlActive = false

    for _, obj in ipairs(swirlParts) do
        if obj and obj.Parent and swirlOriginalCFrames[obj] then
            obj.CFrame = swirlOriginalCFrames[obj]
        end
    end

    swirlParts = {}
    swirlData = {}
    swirlOriginalCFrames = {}
end

local function startPlayerFloat()
    if floatStarted then
        return
    end

    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")

    if not root or not monsterWorldPosition then
        return
    end

    floatStarted = true
    floatStartCFrame = root.CFrame

    local targetPosition = monsterWorldPosition + Vector3.new(0, 40, 0)
    local lookPosition = monsterWorldPosition + Vector3.new(0, 50, 0)

    floatTargetCFrame = CFrame.lookAt(targetPosition, lookPosition)

    pcall(function()
        root.Anchored = true
    end)
end

local function stopPlayerFloat()
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")

    if root then
        pcall(function()
            root.Anchored = false
        end)

        if floatStartCFrame then
            root.CFrame = floatStartCFrame
        end
    end

    floatStarted = false
    floatStartCFrame = nil
    floatTargetCFrame = nil
end

local function updatePlayerFloat()
    if not floatStarted or not floatTargetCFrame then
        return
    end

    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")

    if not root then
        return
    end

    local elapsed = tick() - effectStartTime - 2
    local alpha = math.clamp(elapsed / floatDuration, 0, 1)
    local smoothAlpha = alpha * alpha * (3 - 2 * alpha)

    local targetPosition = monsterWorldPosition + Vector3.new(0, 40, 0)
    local startPosition = floatStartCFrame.Position

    local position = startPosition:Lerp(targetPosition, smoothAlpha)
    local lookPosition = monsterWorldPosition + Vector3.new(0, 50, 0)

    local swayX = math.sin(tick() * 4.5) * 1.5
    local swayY = math.sin(tick() * 3.2) * 1.2
    local swayZ = math.cos(tick() * 3.8) * 1.5

    root.CFrame = CFrame.lookAt(
        position + Vector3.new(swayX, swayY, swayZ),
        lookPosition
    )
end

local function restoreOriginalState()
    for _, obj in ipairs(floorParts) do
        if obj and obj.Parent then
            if originalColors[obj] then
                obj.Color = originalColors[obj]
            end

            if originalCFrames[obj] then
                obj.CFrame = originalCFrames[obj]
            end
        end
    end

    floorParts = {}
    originalColors = {}
    originalCFrames = {}
end

local function updateButtonColor()
    if isActive then
        toggleButton.TextColor3 = Color3.fromRGB(255, 0, 0)
        buttonStroke.Color = Color3.fromRGB(255, 0, 0)
    else
        toggleButton.TextColor3 = Color3.fromRGB(255, 220, 0)
        buttonStroke.Color = Color3.fromRGB(255, 170, 0)
    end
end

local function activateLighting()
    Lighting.Brightness = 0.35
    Lighting.Ambient = Color3.fromRGB(80, 0, 0)
    Lighting.OutdoorAmbient = Color3.fromRGB(55, 0, 0)
    Lighting.ColorShift_Top = Color3.fromRGB(130, 0, 0)
    Lighting.ColorShift_Bottom = Color3.fromRGB(70, 0, 0)
    Lighting.FogColor = Color3.fromRGB(65, 0, 0)
    Lighting.FogStart = 80
    Lighting.FogEnd = 900
end

local function restoreLighting()
    Lighting.Brightness = originalLighting.Brightness
    Lighting.Ambient = originalLighting.Ambient
    Lighting.OutdoorAmbient = originalLighting.OutdoorAmbient
    Lighting.ColorShift_Top = originalLighting.ColorShift_Top
    Lighting.ColorShift_Bottom = originalLighting.ColorShift_Bottom
    Lighting.FogColor = originalLighting.FogColor
    Lighting.FogStart = originalLighting.FogStart
    Lighting.FogEnd = originalLighting.FogEnd
end

local colorSequence = {
    Color3.fromRGB(35, 0, 0),
    Color3.fromRGB(90, 0, 0),
    Color3.fromRGB(150, 0, 0),
    Color3.fromRGB(255, 0, 0),
    Color3.fromRGB(75, 0, 0)
}

local renderConnection

renderConnection = RunService.RenderStepped:Connect(function(dt)
    if not isActive then
        return
    end

    local now = tick()
    local elapsed = now - effectStartTime

    local colorIndex = (now * 18) % #colorSequence + 1
    local targetColor = colorSequence[math.floor(colorIndex)]

    for _, obj in ipairs(floorParts) do
        if obj and obj.Parent then
            obj.Color = targetColor

            local origCF = originalCFrames[obj]

            if origCF and not constructionSwirlActive then
                local offsetX = math.random(-350, 350) / 100
                local offsetZ = math.random(-350, 350) / 100
                local offsetY = math.random(-150, 150) / 100

                obj.CFrame = origCF + Vector3.new(offsetX, offsetY, offsetZ)
            end
        end
    end

    if constructionSwirlActive and monsterWorldPosition then
        local swirlTime = elapsed

        for _, obj in ipairs(swirlParts) do
            if obj and obj.Parent then
                local data = swirlData[obj]

                if data then
                    local angle = data.angle + swirlTime * data.speed * data.spin
                    local radiusPulse = math.sin(swirlTime * 2.5 + data.phase) * 8
                    local radius = data.radius + radiusPulse

                    local x = math.cos(angle) * radius
                    local z = math.sin(angle) * radius
                    local y = data.height + math.sin(swirlTime * 3 + data.phase) * 8

                    local target = monsterWorldPosition + Vector3.new(x, y, z)
                    local original = swirlOriginalCFrames[obj]

                    obj.CFrame =
                        CFrame.lookAt(target, monsterWorldPosition)
                        * CFrame.Angles(
                            math.sin(swirlTime * 2 + data.phase) * 0.7,
                            angle,
                            math.cos(swirlTime * 2.5 + data.phase) * 0.7
                        )
                end
            end
        end
    end

    if elapsed >= 2 and not floatStarted then
        stopConstructionSwirl()
        startPlayerFloat()
    end

    updatePlayerFloat()

    if monster and monster.Parent then
        local shake = math.sin(now * 5) * 0.8
        local float = math.sin(now * 1.7) * 2
        local rotation = math.sin(now * 1.2) * 0.035

        for _, part in ipairs(monsterParts) do
            if part and part.Parent then
                local baseCF = monsterBaseCFrames[part]

                if baseCF then
                    part.CFrame =
                        baseCF
                        * CFrame.new(shake, float, 0)
                        * CFrame.Angles(0, rotation, 0)
                end
            end
        end

        if monsterLight then
            monsterLight.Brightness = 12 + math.abs(math.sin(now * 4)) * 22
        end
    end

    local camera = Workspace.CurrentCamera

    if camera then
        local earthquakeMultiplier = elapsed < 2 and 1.5 or 0.8
        local earthquakeX = math.sin(now * earthquakeSpeed) * earthquakeStrength * earthquakeMultiplier
        local earthquakeY = math.sin(now * (earthquakeSpeed * 1.37)) * earthquakeStrength * earthquakeMultiplier
        local earthquakeZ = math.cos(now * (earthquakeSpeed * 0.83)) * earthquakeStrength * earthquakeMultiplier

        camera.CFrame =
            camera.CFrame
            * CFrame.new(earthquakeX, earthquakeY, earthquakeZ)
            * CFrame.Angles(
                earthquakeY * 0.025,
                earthquakeX * 0.025,
                earthquakeZ * 0.025
            )
    end
end)

toggleButton.MouseButton1Click:Connect(function()
    isActive = not isActive
    updateButtonColor()

    if isActive then
        effectStartTime = tick()
        floatStarted = false
        floatStartCFrame = nil
        floatTargetCFrame = nil

        updateFloorList()
        activateLighting()
        createMonster()
        prepareConstructionSwirl()

        pcall(function()
            sound:Play()
        end)
    else
        stopConstructionSwirl()
        stopPlayerFloat()
        restoreOriginalState()
        restoreLighting()
        destroyMonster()

        pcall(function()
            sound:Stop()
        end)
    end
end)

screenGui.AncestryChanged:Connect(function()
    if not screenGui.Parent then
        if renderConnection then
            renderConnection:Disconnect()
        end

        stopConstructionSwirl()
        stopPlayerFloat()

        pcall(function()
            sound:Stop()
        end)

        restoreOriginalState()
        restoreLighting()
        destroyMonster()
    end
end)
